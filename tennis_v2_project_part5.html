    "logistic regression": (LogisticRegression(max_iter=1000), False),
    "decision tree": (DecisionTreeClassifier(max_depth=5, random_state=42), False),

rows = []
for name, (m, sc) in models.items():

Upgrade the app

Replace app.py with the pick-two-players predictor.

This is the whole app. You (or your AI assistant) can extend it with nicer visuals afterward, but this works end to end.

import streamlit as st
import joblib, json

model = joblib.load("app_model.joblib")
players = json.load(open("player_elo.json"))
names = sorted(players)

st.title("ATP Match Predictor (v2, Elo-powered)")
st.write("Pick two players and a surface. The model uses their Elo ratings to predict the winner.")

a = st.selectbox("Player A", names, index=names.index("Novak Djokovic"))
b = st.selectbox("Player B", names, index=names.index("Rafael Nadal"))
surface = st.selectbox("Surface", ["Hard", "Clay", "Grass"])

pa, pb = players[a], players[b]
elo_diff = pa["elo"] - pb["elo"]
surface_elo_diff = pa[surface] - pb[surface]
prob_a = model.predict_proba([[elo_diff, surface_elo_diff]])[0][1]

st.metric(f"{a} win probability", f"{prob_a*100:.0f}%")
st.progress(prob_a)
st.caption(f"{a} favored" if prob_a > 0.5 else f"{b} favored")
